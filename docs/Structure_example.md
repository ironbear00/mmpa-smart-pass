CREATE OR REPLACE TRIGGER "COMM"."TRG_NFPIS_GOODCODE" 
BEFORE INSERT OR UPDATE OR DELETE ON TC_ITEM_MASTER
REFERENCING OLD AS OLD NEW AS NEW
FOR EACH ROW
BEGIN	
	IF INSERTING THEN 
		INSERT INTO GOODCODE@ORA_NFPIS
		( 
			 GOODCD -- 품종코드
			,GOODKNM -- 품종명국문
			,GOODENM -- 품종명영문
			,GOODSNM --품종학명
			,GOODABB -- 품종약칭
			,GOODUNIT -- 품종수량단위
			,HSKCD -- HSK 코드
			,BIGO -- 비고
			--,AIM -- 정밀검사기간유효코드
			--,AIMCD -- 어류구분코드
			,ERASURE_DATE -- 삭제일자
			,MODIFY_DATE -- 수정일자
			,USERID -- 등록자
			--,WATER -- 담수구분
			--,SHELLFISH -- 폐류구분
			--,DEEP_SEA -- 수은
			--,FARMING -- 양식산여부
			--,ANTIBIOTIC -- 항생물질여부
			--,PIGMENT -- 수은여부
			--,C_BACILLUS -- 대장균여부
			--,MANUAL - 수동판정구분
			--,S_DIOXIDE -- 이산화항 여부
			--,C_MONOXIDE -- CO 여부
			,G_PURPOSE -- 수출입구분
			--,P_PACKING -- 횟감,포장횟감 구분
			--,C_NUMBER -- 세균수 여부
			--,B_TOXIC -- 복어독구분
			--,B_PRODUCT -- 종분류
			,GOVSTOCKYN -- 정부비축용도여부
		) 
		VALUES 
		( 
			 :NEW.ITEM_CD -- GOODCD
			,:NEW.ITEM_NM_KOR -- GOODKNM
			,:NEW.ITEM_NM_ENG -- GOODENM
			,:NEW.SCIENTIFIC_NM_ENG -- GOODSNM
			,:NEW.ITEM_NM_ABB -- GOODABB
			,:NEW.CNT_UNIT -- GOODUNIT
			,:NEW.HSK_CD -- HSKCD
			,:NEW.REMARK -- BIGO
			--,AIM
			--,AIMCD
			,DECODE(:NEW.USE_YN, 'N', TO_CHAR(SYSDATE, 'YYYYMMDD'), NULL) --ERASURE_DATE
			,TO_CHAR(:NEW.UPD_DT, 'YYYYMMDD') -- MODIFY_DATE
			,:NEW.REG_ID -- USERID
			--,WATER
			--,SHELLFISH
			--,DEEP_SEA
			--,FARMING
			--,ANTIBIOTIC
			--,PIGMENT
			--,C_BACILLUS
			--,MANUAL
			--,S_DIOXIDE
			--,C_MONOXIDE
			,:NEW.EXP_IMP_TYPE -- G_PURPOSE
			--,P_PACKING
			--,C_NUMBER
			--,B_TOXIC
			--,B_PRODUCT
			,:NEW.GOV_SAVE_YN -- GOVSTOCKYN
		);
	ELSIF UPDATING THEN 
		IF :NEW.USE_YN = 'N' THEN
			UPDATE GOODCODE@ORA_NFPIS
               SET ERASURE_DATE = TO_CHAR(SYSDATE, 'YYYYMMDD') --ERASURE_DATE
                  ,MODIFY_DATE = TO_CHAR(SYSDATE, 'YYYYMMDD')
                  --,USERID = :OLD.UPD_ID --USERID
             WHERE GOODCD = :NEW.ITEM_CD;
		ELSIF :NEW.USE_YN != 'N' THEN
            UPDATE GOODCODE@ORA_NFPIS
               SET GOODKNM = :NEW.ITEM_NM_KOR
                  ,GOODENM = :NEW.ITEM_NM_ENG
                  ,GOODSNM = :NEW.SCIENTIFIC_NM_ENG
                  ,GOODABB = :NEW.ITEM_NM_ABB
                  ,GOODUNIT = :NEW.CNT_UNIT
                  ,HSKCD = :NEW.HSK_CD
                  ,BIGO = :NEW.REMARK
                  ,MODIFY_DATE = TO_CHAR(:NEW.UPD_DT, 'YYYYMMDD')
                  --,USERID = :NEW.REG_ID
                  ,G_PURPOSE = :NEW.EXP_IMP_TYPE
                  ,GOVSTOCKYN = :NEW.GOV_SAVE_YN
                  ,ERASURE_DATE = DECODE(:NEW.USE_YN, 'N', TO_CHAR(SYSDATE, 'YYYYMMDD'), NULL)
             WHERE GOODCD = :NEW.ITEM_CD;
		END IF;
	ELSIF DELETING THEN 
            UPDATE GOODCODE@ORA_NFPIS
               SET ERASURE_DATE = TO_CHAR(SYSDATE, 'YYYYMMDD') --ERASURE_DATE
                  ,MODIFY_DATE = TO_CHAR(SYSDATE, 'YYYYMMDD')
                  --,USERID = :OLD.UPD_ID --USERID
             WHERE GOODCD = :NEW.ITEM_CD;

	END IF;
	EXCEPTION WHEN OTHERS THEN RAISE;
END TRG_NFPIS_GOODCODE;
/
